create  directories frontend and backend

use cmd or your terminal to go into the backend folder
# cd backend

create a virtual environment here
# python -m venv .venv

activate the virtual environment (make sure to not use powershell from here)
# .\.venv\Scripts\activate

use pip to install flask, flask_sqlalchemy and flask-cors
# pip install flask flask-sqlalchemy flask-cors

create these files in backend folder
app.py
routes.py 
models.py

configure your app.py to be your app in case
# set FLASK_APP=app.py

and set up the environment
# set FLASK_APP=app.py
# set FLASK_ENV=development
